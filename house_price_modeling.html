Project Overview

This project analyzes residential housing data to understand the factors that influence property values and to build predictive models for house prices and overall property quality.

The workflow includes data preparation, exploratory data analysis, feature engineering, regression modeling, Random Forest modeling, XGBoost classification, class-imbalance handling, model evaluation, and feature importance analysis.

Two predictive problems are addressed:

  1. Predicting house sale prices using regression models.
  2. Classifying houses by overall quality using machine learning.

Data Preparation

The same preprocessing steps are applied to both the training and test datasets.

prepare_data <- function(df) {

  # Remove YearBuilt because YearRemodAdd captures related information
  if ("YearBuilt" %in% names(df)) {
    df$YearBuilt <- NULL
  }

  # Remodel year categories
  df$YearRemodAdd <- cut(
    df$YearRemodAdd,
    breaks = c(1949, 2000, 2011),
    labels = c(
      "on or before 2000",
      "after 2000"
    ),
    include.lowest = TRUE
  )

  # Total rooms categories
  df$TotRmsAbvGrd <- cut(
    df$TotRmsAbvGrd,
    breaks = c(0, 5, 10, 15),
    labels = c(
      "0-5",
      "5-10",
      "10-15"
    ),
    include.lowest = TRUE
  )

  # Bedroom categories
  df$BedroomAbvGr <- cut(
    df$BedroomAbvGr,
    breaks = c(-1, 5, 15),
    labels = c(
      "less than or equal to 5",
      "More than 5"
    ),
    include.lowest = TRUE
  )

  df
}

Evaluation Function

regression_metrics <- function(actual, predicted) {

  mse <- mean(
    (actual - predicted)^2
  )

  rmse <- sqrt(mse)

  r_squared <- cor(
    actual,
    predicted
  )^2

  data.frame(
    MSE = mse,
    RMSE = rmse,
    R_Squared = r_squared
  )
}

XGBoost Helper Functions

This version uses the native XGBoost interface instead of the older caret::train(method = "xgbTree") interface.

make_xgb_matrices <- function(
  training_data,
  testing_data
) {

  train_x <- training_data[
    ,
    names(training_data) != "OverallQual"
  ]

  test_x <- testing_data[
    ,
    names(testing_data) != "OverallQual"
  ]

  combined <- rbind(
    train_x,
    test_x
  )

  combined_matrix <- model.matrix(
    ~ . - 1,
    data = combined
  )

  n_train <- nrow(
    train_x
  )

  x_train <- combined_matrix[
    1:n_train,
    ,
    drop = FALSE
  ]

  x_test <- combined_matrix[
    (n_train + 1):
      nrow(combined_matrix),
    ,
    drop = FALSE
  ]

  list(
    train = x_train,
    test = x_test
  )
}

Classification Evaluation

classification_metrics <- function(
  actual,
  predicted
) {

  actual <- as.integer(
    actual
  )

  predicted <- as.integer(
    predicted
  )

  TP <- sum(
    predicted == 1 &
    actual == 1
  )

  TN <- sum(
    predicted == 0 &
    actual == 0
  )

  FP <- sum(
    predicted == 1 &
    actual == 0
  )

  FN <- sum(
    predicted == 0 &
    actual == 1
  )

  accuracy <-
    (TP + TN) /
    (TP + TN + FP + FN)

  sensitivity <- ifelse(
    TP + FN == 0,
    NA,
    TP / (TP + FN)
  )

  specificity <- ifelse(
    TN + FP == 0,
    NA,
    TN / (TN + FP)
  )

  precision <- ifelse(
    TP + FP == 0,
    NA,
    TP / (TP + FP)
  )

  f1 <- ifelse(
    is.na(precision) |
      is.na(sensitivity) |
      precision + sensitivity == 0,
    NA,
    2 *
      precision *
      sensitivity /
      (precision + sensitivity)
  )

  data.frame(
    Accuracy = accuracy,
    Sensitivity = sensitivity,
    Specificity = specificity,
    Precision = precision,
    F1 = f1
  )
}

XGBoost Training Function

fit_xgb_model <- function(
  training_data,
  testing_data,
  seed = 345
) {

  set.seed(seed)

  matrices <- make_xgb_matrices(
    training_data,
    testing_data
  )

  y_train <- as.numeric(
    training_data$OverallQual
  )

  dtrain <- xgb.DMatrix(
    data = matrices$train,
    label = y_train
  )

  dtest <- xgb.DMatrix(
    data = matrices$test
  )

  params <- list(
    objective = "binary:logistic",
    eval_metric = "logloss",
    max_depth = 6,
    eta = 0.1,
    gamma = 0,
    min_child_weight = 1,
    subsample = 1,
    colsample_bytree = 1
  )

  model <- xgb.train(
    params = params,
    data = dtrain,
    nrounds = 100,
    verbose = 0
  )

  probability <- predict(
    model,
    dtest
  )

  prediction <- ifelse(
    probability >= 0.5,
    1,
    0
  )

  list(
    model = model,
    prediction = prediction,
    probability = probability,
    feature_names = colnames(
      matrices$train
    )
  )
}

Oversampling

The minority class is oversampled to create a more balanced training dataset.

oversample_binary <- function(
  df,
  target = "OverallQual",
  seed = 345
) {

  set.seed(seed)

  class0 <- df[
    df[[target]] == 0,
  ]

  class1 <- df[
    df[[target]] == 1,
  ]

  target_size <- max(
    nrow(class0),
    nrow(class1)
  )

  class0_over <- class0[
    sample(
      seq_len(nrow(class0)),
      target_size,
      replace = TRUE
    ),
  ]

  class1_over <- class1[
    sample(
      seq_len(nrow(class1)),
      target_size,
      replace = TRUE
    ),
  ]

  result <- rbind(
    class0_over,
    class1_over
  )

  result[
    sample(seq_len(nrow(result))),
  ]
}

Key Insights

The analysis demonstrates several important points:

  • House prices depend on a combination of property characteristics rather than a single factor.
  • Lot size, garage area, finished basement area, bathrooms, bedrooms, fireplaces, and overall quality can contribute to property value.
  • Regression and ensemble approaches can be compared using MSE, RMSE, and R-squared.
  • Class imbalance can materially affect classification performance.
  • Oversampling and undersampling provide alternative approaches for improving minority-class prediction.
  • Feature importance helps identify the variables contributing most strongly to model predictions.
